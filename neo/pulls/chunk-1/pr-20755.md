Resolves #20752

Suppresses zero-delta VDOM flights in `Neo.grid.View#syncBodies` by latching `Neo.grid.Body#hasSilentVdomChanges` across silent row and body VDOM writes (`Row#createVdom`, `Body#createRowPool`, deferred `Body#createViewData`, `Body#syncAriaRowOwns`, `Body#updateScrollHeight`, and `Body#updateVdom` under `silentVdomUpdate`), returning early before scheduling a View update when `force` is `false`, `#scrollCorrection` is `0`, and no active body latched a silent mutation, and restoring `hasSilentVdomChanges` when a View or Body flight fails before `afterVnodeSync` lands.

Evidence: L4 (live Chrome 3-worker VDOM pipeline probe on `examples/grid/bigData` + Playwright unit specs) → L4 required (AC-1, AC-2, AC-3, AC-4, AC-5).

- Selected an explicit `Body#hasSilentVdomChanges` latch rather than `createViewData` return booleans or `_needsVdomUpdate` alone: it captures pre-`createViewData` silent mutations (`ScrollManager` `setSilent({isScrolling})`, `maxScrollHeight` / `refreshRowGeometry` `updateScrollHeight(true)`, `rowMeasureVdom`), transfers claimed bodies through `View#getVdomUpdatePayload` / `afterVnodeSync` so scroll ticks inside `executeVdomUpdate` do not schedule duplicate flights, and restores `hasSilentVdomChanges` if a direct, queued, or standalone body flight rejects before landing.

/**
 * Runs in the App worker: counts the view's flights from now on.
 */
function countFlights(viewId) {
    const Helper   = Neo.vdom.Helper,
          original = Helper.updateBatch,
          counter  = globalThis.__review = {flights: 0, zeroDelta: 0, roots: {}};

    const wrapped = function(data) {
        const ids = Object.keys(data.updates);

        return Promise.resolve(original.call(this, data)).then(response => {
            if (ids.includes(viewId)) {
                counter.flights++;
                (response?.deltas?.length ?? 0) === 0 && counter.zeroDelta++
            } else {
                ids.forEach(id => {
                    const key = id.replace(/-?\d+$/, '').replace(/__.*$/, '');

                    counter.roots[key] = (counter.roots[key] || 0) + 1
                })
            }

            return response
        })
    };

    wrapped.remoteProvenance = original.remoteProvenance;
    Helper.updateBatch       = wrapped;

    return true
}

/**
 * Runs in the App worker after the script settled: one dense flight of the view at row depth, and what it found.
 */
async function leftover(viewId) {
    const view  = Neo.getComponent(viewId),
          sleep = ms => new Promise(resolve => setTimeout(resolve, ms));

    for (let i = 0; i < 200 && (view.isVdomUpdating || view.needsVdomUpdate); i++) {
        await sleep(10)
    }

    const counter = {...globalThis.__review},
          latched = [view.bodyStart, view.gridContainer.body, view.bodyEnd].filter(Boolean).map(body => body.hasSilentVdomChanges ?? null);

    view.updateDepth = 3 + view.maxCellDepth;
    view.denseUpdate = true;

    const result = await view.promiseUpdate(),
          kinds  = {};

    (result?.deltas || []).forEach(delta => {
        const key = delta.action || Object.keys(delta).filter(name => name !== 'id').sort().join('+') || 'update';

        kinds[key] = (kinds[key] || 0) + 1
    });

    return {
        counter, latched,
        leftover     : result?.deltas?.length ?? null,
        leftoverKinds: kinds,
        sample       : (result?.deltas || []).slice(0, 3).map(delta => JSON.stringify(delta).slice(0, 160))
    }
}
